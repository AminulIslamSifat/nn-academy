Building a Reusable Layer

class LinearLayer:
    def __init__(self, n_in: int, n_out: int):
        self.W = np.random.randn(n_in, n_out) * np.sqrt(2.0 / n_in)
        self.b = np.zeros(n_out)
    
    def forward(self, X):
        return X @ self.W + self.b
